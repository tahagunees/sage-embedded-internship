cmake_minimum_required(VERSION 3.16)
project(hil_host_tests C)
set(CMAKE_C_STANDARD 11)
add_executable(hil_tests main.c ../test_hil.c ../../main/motor_model.c ../../main/stm_protocol.c)
target_include_directories(hil_tests PRIVATE . ../../main)
target_compile_definitions(hil_tests PRIVATE HIL_HOST_TEST=1)
if(MSVC)
    target_compile_options(hil_tests PRIVATE /W4 /WX)
else()
    target_compile_options(hil_tests PRIVATE -Wall -Wextra -Werror -pedantic)
    target_link_libraries(hil_tests PRIVATE m)
endif()
enable_testing()
add_test(NAME hil_portable_tests COMMAND hil_tests)
